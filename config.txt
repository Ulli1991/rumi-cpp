# =============================================================================
# C++ N-Body Configuration
# =============================================================================

# Test case: binary_test, binary_test_peri, triple_test, outer_planets, plummer
TEST_CASE = triple_test

# Integrator: fsi (global timestep) or hhs (adaptive HHS-FSI per Algorithm 4 of FROST paper)
INTEGRATOR = hhs

# =============================================================================
# Simulation parameters
# =============================================================================
TOTAL_TIME_MYR = 2.5e-7                    	# ~1000 orbits for BH binary at 0.01 AU
DT_MAX_MYR = 1.0e-9                       	# Must be small initially; LogH takes over after detection
DT_MIN_MYR = 1.0e-14                      	# Needs to resolve periapsis before subsystem detected
ETA = 0.2					# Integrator accuracy eta
SOFTENING_PC = 1e-3                        	# Softening length in pc (= 206 AU)
SERIAL_N2_GPU = 16384                      	# Full N² on GPU without MPI or tree below this count
SERIAL_N2_CPU = 256                        	# Full N² on CPU below this count (avoids kernel launch overhead)
FUSE_GRADIENT = false                      	# Fuse force+gradient into single N^2 pass using stale accelerations (saves ~30% gradient time, O(dt^3) accuracy trade-off)
INTEGER_POSITIONS = false                  	# Use int64 positions (exact drift, MPI-safe)
DOMAIN_HALFSIZE_PC = 200                   	# Integer position domain half-size in pc
MPI_NON_BLOCKING = false                    	# Use MPI_Iallgatherv for overlapped transfers
SYMMETRIC_TIMESTEPS = true                	# Fuse force+timestep with harmonic symmetrization (FROST paper Eq. 23)

# =============================================================================
# Barnes-Hut tree: O(N log N) force via monopole approximation
# =============================================================================
USE_TREE = false                           	# Master switch to enable tree for self-forces
TREE_THRESHOLD = 100000 	                # Use tree above this count, N² below
TREE_THETA = 0.7                          	# Opening angle (smaller = more accurate, slower)
TREE_N_LEAF = 32                          	# Max particles per leaf cell
TREE_N_GROUP = 64                         	# Particles per traversal group
GADGET4_CROSS = false                     	# F_cross = F_all - F_self decomposition (requires USE_TREE)

# =============================================================================
# Collisional dynamics
# Enable for unsoftened direct N-body with close encounter handling.
# Forces SOFTENING_PC = 0 when enabled.
# =============================================================================
COLLISIONAL = true 				# Master switch to enable proper collional treatment of the Nbody system
R_NGB = 1e-3                        		# Neighbor search radius (pc) for subsystem detection
R_NGB_MAX = 0.1                     		# Max distance for orbital element check (pc)
ETA_GBS = 1e-10                     		# GBS extrapolation accuracy for LogH integrator
N_ORB_SECULAR = 3                   		# Orbits/timestep threshold for secular evolution
M_PN = 5.0                          		# Minimum mass (M_sun) for PN corrections
PN_ORDER = 31                        		# PN bitmask: 1=PN1, 2=PN2, 4=PN2.5 (7=all up to PN2.5)
R_ESCAPE_RH = 100.0                 		# Escape radius in half-mass radii
MSTAR_THRESHOLD = 8                 		# N-body threshold: MSTAR (>=) vs LogH (<)
SSE_ENABLED = false                  		# Runtime single stellar evolution
SSE_METALLICITY = 0.02              		# Metallicity Z for SSE

# =============================================================================
# Binary test parameters (only used if TEST_CASE = binary_test)
# BH binary for PN precession test (BIFROST Fig. 2).
# Period ~ 2.24e-4 yr at 0.01 AU. PN1 precession ~ 2e-3 rad/orbit.
# =============================================================================
BINARY_A_PC = 1e-8				# Semi-major axis in pc (= 0.01 AU, 4.848e-8)
BINARY_E = 0.9					# Eccentricity (high for visible precession)
BINARY_M1 = 10.0				# Primary mass in M_sun (BH)
BINARY_M2 = 10.0				# Secondary mass in M_sun (BH)

# =============================================================================
# Triple test parameters (only used if TEST_CASE = triple_test)
# Hierarchical triple: inner binary (m1,m2) + outer body m3.
# Default: 3x10 M_sun BHs, a_out/a_in=20, i=85° for Kozai-Lidov.
# Kozai-Lidov timescale ~ (M_tot/m3)(a_out/a_in)^3 * P_in / (1-e_out^2)^{3/2}
# =============================================================================
TRIPLE_A_IN_PC = 1e-8				# Inner semi-major axis in pc
TRIPLE_E_IN = 0.1				# Inner eccentricity
TRIPLE_M1 = 10.0				# Inner primary mass (M_sun)
TRIPLE_M2 = 10.0				# Inner secondary mass (M_sun)
TRIPLE_A_OUT_PC = 4e-8				# Outer semi-major axis in pc
TRIPLE_E_OUT = 0.7				# Outer eccentricity
TRIPLE_M3 = 10.0				# Outer body mass (M_sun)
TRIPLE_INCL_DEG = 85.0				# Mutual inclination (degrees); >39 for Kozai-Lidov
# To test unstable triple (LogH 3-body, no secular): set TRIPLE_A_OUT_PC = 4e-8
# (a_out/a_in=4) and TRIPLE_E_OUT = 0.7. This fails Mardling → TRI_U → full LogH
# integration of the chaotic 3-body interaction (much slower, tests close encounters).

# =============================================================================
# Plummer cluster parameters (only used if TEST_CASE = plummer)
# =============================================================================
PLUMMER_N = 1000000
PLUMMER_M_TOTAL = 3.91e5            		# Total mass in solar masses (not used when kroupa_evolved)
PLUMMER_R_HALF = 3.5                		# Half-mass radius in parsecs
PLUMMER_SEED = 42                               # random seed for the mass sampling (I like 42)
PLUMMER_MASS_DIST = kroupa_evolved  		# equal, kroupa, or kroupa_evolved
PLUMMER_AGE_GYR = 1.0               		# Cluster age in Gyr (for kroupa_evolved)
PLUMMER_METALLICITY = 0.02          		# Stellar metallicity Z (solar)

# =============================================================================
# Output
# =============================================================================
PRINT_INTERVAL = 1
PROFILING = true                    		# Print performance breakdown at each print interval
