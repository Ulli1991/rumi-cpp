# =============================================================================

# =============================================================================
# Triple test parameters (only used if TEST_CASE = triple_test)
# Hierarchical triple: inner binary (m1,m2) + outer body m3.
# Default: 3x10 M_sun BHs, a_out/a_in=20, i=85° for Kozai-Lidov.
# Kozai-Lidov timescale ~ (M_tot/m3)(a_out/a_in)^3 * P_in / (1-e_out^2)^{3/2}
# =============================================================================
TRIPLE_A_IN_PC = 1e-8				# Inner semi-major axis in pc
TRIPLE_E_IN = 0.1				# Inner eccentricity
TRIPLE_M1 = 10.0				# Inner primary mass (M_sun)

TRIPLE_A_OUT_PC = 5e-8				# Outer semi-major axis in pc (a_out/a_in=5)
TRIPLE_E_OUT = 0.1				# Outer eccentricity
TRIPLE_M3 = 10.0				# Outer body mass (M_sun)
TRIPLE_INCL_DEG = 45.0				# Mutual inclination (degrees); >39 for Kozai-Lidov
# Clean unstable triple: a_out/a_in=5, e_out=0.1, i=45°. Clearly Mardling-unstable
# but mild Kozai (e_max~0.47) — tests LogH 3-body without PN blowup.
#
# Hard case (Kozai-driven PN blowup): TRIPLE_A_OUT_PC=4e-8, TRIPLE_E_OUT=0.7,
# TRIPLE_INCL_DEG=85.0 → a_out/a_in=4, near-resonant Kozai, drives e→1 and B→0.
